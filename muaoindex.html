<script>

const canvas = document.getElementById("gameCanvas");

const ctx = canvas.getContext("2d");


canvas.width = 800;

canvas.height = 600;


let score = 0;

let lives = 3;

let gameOver = false;


// Detetor do Jogador

const detector = {

    x: canvas.width / 2 - 50,

    y: canvas.height - 40,

    width: 100,

    height: 20,

    speed: 8

};


// Lista de Muões no ecrã

let muons = [];


// Controlos

let keys = {};

window.addEventListener("keydown", e => keys[e.key] = true);

window.addEventListener("keyup", e => keys[e.key] = false);


// Função para criar um novo Muão

function spawnMuon() {

    if (gameOver) return;

    muons.push({

        x: Math.random() * (canvas.width - 20) + 10,

        y: 0,

        radius: 10,

        speed: Math.random() * 2 + 2,

        // O "tempo de vida" do muão simulado por uma taxa de desintegração aleatória

        lifespan: 1.0 

    });

    // Criar um novo muão a cada 1 segundo

    setTimeout(spawnMuon, 1000);

}


// Atualizar a lógica do jogo

function update() {

    if (gameOver) return;


    // Mover Detetor

    if (keys["ArrowLeft"] && detector.x > 0) detector.x -= detector.speed;

    if (keys["ArrowRight"] && detector.x < canvas.width - detector.width) detector.x += detector.speed;


    // Atualizar Muões

    for (let i = muons.length - 1; i >= 0; i--) {

        let m = muons[i];

        m.y += m.speed;

        m.lifespan -= 0.0015; // O muão vai decaindo com o tempo


        // Se o muão se desintegrar antes de chegar ao chão

        if (m.lifespan <= 0) {

            muons.splice(i, 1);

            continue;

        }


        // Colisão com o Detetor

        if (m.y + m.radius >= detector.y && m.x >= detector.x && m.x <= detector.x + detector.width) {

            score += 10;

            document.getElementById("score").innerText = score;

            muons.splice(i, 1);

            continue;

        }


        // Se passar o detetor e tocar no chão (o cientista perdeu a oportunidade de detetá-lo)

        if (m.y > canvas.height) {

            muons.splice(i, 1);

            lives--;

            document.getElementById("lives").innerText = lives;

            if (lives <= 0) {

                gameOver = true;

                alert("Fim de Jogo! Conseguiu detetar " + score + " pontos de muões.");

            }

        }

    }

}


// Desenhar os elementos no ecrã

function draw() {

    ctx.clearRect(0, 0, canvas.width, canvas.height);


    // Desenhar Detetor

    ctx.fillStyle = "#00ffcc";

    ctx.fillRect(detector.x, detector.y, detector.width, detector.height);


    // Desenhar Muões

    muons.forEach(m => {

        ctx.beginPath();

        ctx.arc(m.x, m.y, m.radius, 0, Math.PI * 2);

        // A cor muda (fica mais fraca) conforme o muão se aproxima da desintegração

        ctx.fillStyle = `rgba(255, 50, 50, ${m.lifespan})`;

        ctx.fill();

        ctx.closePath();

    });

}


// Loop do Jogo

function gameLoop() {

    update();

    draw();

    requestAnimationFrame(gameLoop);

}


// Iniciar

spawnMuon();

gameLoop();

</script>
